\section{Several Rings: Transfer of Moments and Confinement}
\label{sec:several}

We bound the velocity fields that the rings of $\omega^{(n)}$ induce on one
another, and we show that, on a time window after ring $k+1$ saturates, the
material inside ring $k$ stays in the disk $\overline B_{ar_k/8}$.

We assume \textup{(N1)}--\textup{(N3)} of Hypothesis~I.\ref{I-hyp:N}, fix
$n\ge2$, and use the datum and its approximations $\omega^{(n)}$, the ring
quantities and the clock of Definitions~\ref{def:datum}, \ref{def:approx},
\ref{def:regions} and \ref{def:clock}, with $\mu\in(1,2]$ and $\sbar\in(0,a/16]$; we write
$\omega=\omega^{(n)}$, $v=\KBS\ast\omega$, and $\flow_t$ for the flow of $v$.
Since $s_k=\sbar(2\mu)^{-k}\le\sbar/2$ and $r_k/r_i=s_is_{i+1}\cdots s_{k-1}$,
\begin{equation}
\label{eq:sk-small}
s_k\le\frac a{32}\quad(k\ge1),\qquad
\frac{r_k}{r_i}\le s_{k-1}\,\sbar^{\,k-i-1}\quad(1\le i<k).
\end{equation}
Moreover $\sbar<\frac12$ and
$\mu^4\sbar^2\le16\cdot2^{-8}<\frac18$, because $a<1$. The conditions
$\sbar\le a/16$, $\sbar\le\frac12$ and $\mu^4\sbar^2\le\frac18$ appear
among the hypotheses of Lemmas~\ref{lem:transfer} and~\ref{lem:field-bound} and of
Proposition~\ref{prop:confine} so that these can be quoted on
their own; under the standing assumptions of this section they always hold.

In this section $q$, without an index, is the order of a derivative.
For a velocity field $V$, $V_{[j]}$ is $V$ in ring-$j$ units,
that is, $V_{[j]}(y)=V(r_jy)/(A_jr_j)$ (defined before
Definition~\ref{def:regions}). As in
Section~I.\ref{I-sec:prelim}, $\|h\|_{C^k(U)}$ is the sum over $q\le k$ of
$\sup_U|\nabla^qh|$, so the proofs below bound each $\nabla^q$ and then sum over
$q$. We use the \textbf{scaling
rule}: if $c\in\R$, $c'>0$ and $W(y)=cV(c'y)$, then
$\nabla^qW(y)=c\,c'^{\,q}(\nabla^qV)(c'y)$.

In the complex-variable calculations below, we use the spatial variable
$z\in\C$ and the conventions for $V(z)$ and its Cartesian derivatives
from Section~\ref{sec:II-prelim}.
We write $F(z)=V_1(z)-iV_2(z)$ for the complex form of a vector field $V$, and we use
Equation~\eqref{eq:BS-complex} for the complex form of $\KBS\ast g$
(Section~\ref{sec:II-prelim}). The monopole
$\frac1{2\pi}\big(\int g\big)(-\Im z,\Re z)/|z|^2$ of $g$ has complex form $\big(\int g\big)/(2\pi iz)$,
and the rotation $\Orot$ by the angle $2\pi/3$ (Section~I.\ref{I-sec:prelim}) is
multiplication by $e^{2\pi i/3}$. For real $g$, the complex forms
of $\KBS\ast g$ and of $\KBS\ast g$ minus its monopole are holomorphic on
$U=\C\setminus(\supp g\cup\{0\})$, so Equation~\eqref{eq:BS-complex-deriv}
applies to them there.

\subsection{Fields Induced by One Ring on Another}

Off the support of a function $g$, the complex form of the velocity of $g$ is
given by Equation~\eqref{eq:BS-complex}, and expanding $(z-\zeta)^{-1}$ turns it
into a power series: in $z^{-1}$ outside a disk centered at the origin that
contains the support, and in $z$ inside such a disk that does not meet it. For a $3$-fold rotationally
symmetric $g$ only every third coefficient survives. For material inside
ring $j$, apart from the monopole, the series starts at
$z^{-4}$, and on $N'_j$ this gives the factor $\varepsilon_j=\mu s_j^5$ of
Definition~\ref{def:regions} in parts (1)--(3) of the next lemma; for material
outside ring $k$ it starts at $z^2$, which gives the factor $b_{i,k}$ in
part (4).

\begin{lemma}
\label{lem:transfer}
Assume $\sbar\le a/16$, and let $M_h\ge1$ and $s'\ge0$ be integers. There is
$\Cvel\ge1$, depending only on $a$ and $M_h$, such that the following hold at
every time $t\ge0$, for $1\le j<n$. We choose this constant large enough for
Lemmas~\ref{lem:inner-field} and~\ref{lem:field-bound} below as well.
\begin{enumerate}[label=\textup{(\arabic*)}]
\item Let $g\in L^1(\R^2)$ be real valued, $3$-fold rotationally symmetric and supported in
  $\overline B_{ar_{j+1}/8}$, and let $E_g$ be its velocity field minus its
  monopole. Then
  \[
  \|(E_g)_{[j]}\|_{C^{M_h}(N')}
  \le \Cvel\varepsilon_j\sup_{|y|=a/4}|(E_g)_{[j+1]}(y)|.
  \]
\item If $S_{j+1}(t)\subset N_{j+1}$, let
  $V=\KBS\ast\big(A_{j+1}w_{j+1}(t,\cdot/r_{j+1})\big)$. Then
  \[
  \|V_{[j]}\|_{C^{M_h}(N')}
  \le \Cvel\varepsilon_j\|w_{j+1}(t)\|_{H^{s'}}.
  \]
\item If $D_{j+1}(t)\subset\overline B_R$ with $R\le ar_j/8$, let $E$ be the
  velocity field of $\omega_{\ge j+1}(t)$ minus its monopole. Then
  \[
  \|E_{[j]}\|_{C^{M_h}(N')}
  \le \Cvel\varepsilon_j(R/r_{j+1})^3.
  \]
\item For $1\le i<k\le n$ with $S_i(t)\subset N_i$, put
  $V_i=\KBS\ast\big(A_iw_i(t,\cdot/r_i)\big)$ and
  $b_{i,k}=\mu^{i-k}r_k/r_i$. Then
  \[
  \begin{aligned}
  \|(V_i)_{[k]}\|_{C^{M_h}(N')}&\le \Cvel b_{i,k}\|w_i(t)\|_{L^1},\\
  \|(V_i)_{[k]}\|_{C^{M_h}(N')}&\le \Cvel b_{i,k}\|w_i(t)\|_{H^{s'}}.
  \end{aligned}
  \]
  If $S_i(t)\subset N_i$ for every $i<k$, put
  $V_{\mathrm{out}}=\KBS\ast\big(\omega(t)\mathbf 1_{S_1(t)\cup\dots\cup S_{k-1}(t)}\big)$.
  On $N'_k$, $V_{\mathrm{out}}=\sum_{i<k}V_i$, and
  \[
  \|(V_{\mathrm{out}})_{[k]}\|_{C^{M_h}(N')}
  \le \Cvel\sum_{i<k}b_{i,k}\|w_i(t)\|_{L^1}.
  \]
\end{enumerate}
\end{lemma}

\begin{proof}
Throughout this proof, implicit constants depend only on $a$ and $M_h$.
The differentiated geometric series are finite:
\[
\sum_{p\ge1}(3p+q)^q2^{-3(p-1)}<\infty
\qquad(0\le q\le M_h).
\]
We prove the four bounds with implicit constants, which are absorbed into
$\Cvel$.
The $H^{s'}$ bounds then follow from
$\|w\|_{L^1}\le |N|^{1/2}\|w\|_{L^2}\le4\|w\|_{H^{s'}}$,
since each perturbation is supported in $N\subset\overline B_2$
(Lemma~\ref{lem:regions-basic}(4)). Put $\rho_0=a/4$, the inner radius of $N'$.

\emph{Proof of (1).} We bound the coefficients of the series of the field of
$g$ minus its monopole, in ring-$(j+1)$ units, by the values of this field on the circle $|z|=\rho_0$, and then sum
the series at the points of $N'_j$, which in these units lie outside
$B_{\rho_0/s_j}$. Put $V=(E_g)_{[j+1]}$,
$\mathcal M=\sup_{|z|=\rho_0}|V(z)|$ and
$\tilde g(\zeta)=g(r_{j+1}\zeta)/A_{j+1}$. Since $\KBS$ is homogeneous of degree
$-1$, $V$ is the field of $\tilde g$ minus its monopole, and $\tilde g$ is
$3$-fold rotationally symmetric and supported in $\overline B_{a/8}$. Inserting
$(z-\zeta)^{-1}=\sum_{l\ge0}\zeta^lz^{-l-1}$, uniformly convergent for
$|\zeta|\le a/8$ and $z$ in compact subsets of $\{|z|>a/8\}$, in
Equation~\eqref{eq:BS-complex} and integrating term by term, we find that the
complex form of $V$ is
\[
F(z)=\sum_{l\ge1}\alpha_lz^{-l-1},\qquad \alpha_l=\frac1{2\pi i}\int\tilde g(\zeta)\zeta^l\dd A(\zeta)\qquad(|z|>a/8),
\]
without a term $l=0$: in the expansion of the full field that term is the
monopole, which $V$ does not contain. The substitution $\zeta=\Orot\zeta'$ and the $3$-fold
rotational symmetry of $\tilde g$ give
$\alpha_l=e^{2\pi il/3}\alpha_l$, so $\alpha_l=0$ unless $3\mid l$. On
$|z|=\rho_0$ the series converges uniformly, and $\alpha_l\rho_0^{-l-1}$ is the
Fourier coefficient of $\theta\mapsto F(\rho_0e^{i\theta})$ at the frequency
$-(l+1)$; since $V$ is real, $|F|=|V|$, so $|\alpha_l|\le\mathcal M\rho_0^{l+1}$. For
$|z|\ge\rho_0/s_j$, since $(l+1)\cdots(l+q)\le(l+q)^q$ and $s_j\le\frac12$
(Equation~\eqref{eq:sk-small}),
\begin{align*}
|F^{(q)}(z)|&\le\mathcal M\sum_{l\in 3\N}(l+q)^q\rho_0^{l+1}\Big(\frac{s_j}{\rho_0}\Big)^{l+1+q}
=\mathcal M\rho_0^{-q}s_j^{4+q}\sum_{p\ge1}(3p+q)^qs_j^{3(p-1)}\\
&\le\mathcal M\Big(\frac4a\Big)^qC\,s_j^{4+q}.
\end{align*}
In ring-$j$ units
$(E_g)_{[j]}(z)=\mu s_jV(z/s_j)$, since $A_{j+1}r_{j+1}=\mu s_jA_jr_j$, and
$|z/s_j|\ge\rho_0/s_j$ for $z\in N'$. By the scaling rule and
Equation~\eqref{eq:BS-complex-deriv}, on $N'$,
$|\nabla^q(E_g)_{[j]}(z)|=\mu s_j^{1-q}2^{q/2}|F^{(q)}(z/s_j)|$, which is at most
$C\mu s_j^5\mathcal M$. Summing over $q\le M_h$ gives
$\|(E_g)_{[j]}\|_{C^{M_h}(N')}\lesssim\varepsilon_j\mathcal M$,
which gives (1).

\emph{Proof of (2).} Recall that $w_{j+1}(t,\cdot)$ is already expressed in
ring-$(j+1)$ spatial and vorticity units; its first argument $t$ is physical
time. The corresponding perturbation in physical coordinates is
$A_{j+1}w_{j+1}(t,\cdot/r_{j+1})$, whose velocity field is the $V$ of the
statement. By homogeneity,
\[
V_{[j+1]}(z)=\big(\KBS\ast w_{j+1}(t,\cdot)\big)(z).
\]
By Lemma~\ref{lem:regions-basic}(4), $w_{j+1}(t,\cdot)$ is $3$-fold
rotationally symmetric, supported in $N\subset\overline B_2$, and has zero
integral. As in (1), now with $w_{j+1}(t,\cdot)$ in place of $\tilde g$, the
complex form of $V_{[j+1]}$ is
$F(z)=\sum_{l\in 3\N}\alpha_lz^{-l-1}$ for $|z|>2$. Here
$|\alpha_l|\le\frac1{2\pi}2^l\|w_{j+1}(t)\|_{L^1}$, because $|\zeta|\le2$ on $N$,
and the zero integral removes the term $l=0$.
For $|z|\ge\rho_0/s_j$ we have $2/|z|\le8s_j/a\le\frac14$ by
Equation~\eqref{eq:sk-small}, and
\[
\begin{aligned}
|F^{(q)}(z)|&\le\frac{\|w_{j+1}(t)\|_{L^1}}{2\pi}\sum_{l\in 3\N}(l+q)^q2^l\Big(\frac{4s_j}a\Big)^{l+1+q}\\
&\le\frac{\|w_{j+1}(t)\|_{L^1}}{2\pi}\Big(\frac4a\Big)^{1+q}\Big(\frac8a\Big)^3C\,s_j^{4+q}.
\end{aligned}
\]
The second inequality follows, as in (1), by writing $l=3p$ and using
$8s_j/a\le\frac12$. Passing to ring-$j$ units as in (1),
$|\nabla^qV_{[j]}(y)|\lesssim\varepsilon_j\|w_{j+1}(t)\|_{L^1}$
on $N'$, and summing over $q\le M_h$ proves (2).

\emph{Proof of (3).} By Lemma~\ref{lem:regions-basic}(3), $g=\omega_{\ge j+1}(t)$
is $3$-fold rotationally symmetric and supported in $D_{j+1}(t)\subset\overline B_R$, and by
Lemma~\ref{lem:speed}, $\|g\|_{L^1}\le2\|\Phi\|_{L^1}A_{j+1}r_{j+1}^2\le2\pi
A_{j+1}r_{j+1}^2$, since $0\le\Phi<1$ and $\supp\Phi\subset\overline B_1$. Every
$x\in N'_j$ has $|x|\ge ar_j/4\ge2R$. For the field $E$ of $g$ minus its
monopole, Lemma~\ref{lem:ext-multipole} with $\varrho=R$ and the scaling rule
(with $c=1/(A_jr_j)$ and $c'=r_j$) give, for $y\in N'$,
\begin{align*}
|\nabla^qE_{[j]}(y)|&=\frac{r_j^{q}}{A_jr_j}|\nabla^qE(r_jy)|
\le C_{3,q}\frac{2\pi A_{j+1}r_{j+1}^2R^3}{A_jr_j^5}|y|^{-4-q}\\
&\le2\pi C_{3,q}\Big(\frac4a\Big)^{4+q}\mu s_j^2\Big(\frac R{r_j}\Big)^3 .
\end{align*}
The second inequality uses $|y|\ge a/4$ on $N'$ and
$A_{j+1}r_{j+1}^2=\mu s_j^2A_jr_j^2$. Since $\mu s_j^2(R/r_j)^3=\mu s_j^5(R/r_{j+1})^3=\varepsilon_j(R/r_{j+1})^3$,
summing over $q\le M_h$ gives the bound
$\lesssim\varepsilon_j(R/r_{j+1})^3$, proving (3).

\emph{Proof of (4).} Now the field is generated by a ring outside the annulus
on which we evaluate it, so we expand it in nonnegative powers of $z$, in
ring-$i$ units. Recall that $w_i(t,\cdot)$ is already expressed in ring-$i$
spatial and vorticity units, while $t$ is physical time. The perturbation in
physical coordinates is $A_iw_i(t,\cdot/r_i)$, whose velocity field is the
$V_i$ of the statement. Put $s_{i,k}=r_k/r_i\le a/32$
(Equation~\eqref{eq:sk-small}). By homogeneity,
\[
(V_i)_{[i]}(z)=\big(\KBS\ast w_i(t,\cdot)\big)(z).
\]
The function $w_i(t,\cdot)$ is $3$-fold rotationally symmetric and supported in
$N=\{a/2\le|\zeta|\le2\}$. Inserting
$(z-\zeta)^{-1}=-\sum_{l\ge0}z^l\zeta^{-l-1}$, uniformly convergent for
$\zeta\in N$ and $z$ in compact subsets of $B_{a/2}$, in
Equation~\eqref{eq:BS-complex} gives the complex form of $(V_i)_{[i]}$, for $|z|<a/2$:
\[
\begin{gathered}
F(z)=-\sum_{l\ge0}\alpha_lz^l,\qquad
\alpha_l=\frac1{2\pi i}\int w_i(t,\zeta)\zeta^{-l-1}\dd A(\zeta),\\
|\alpha_l|\le\frac{\|w_i(t)\|_{L^1}}{2\pi}\Big(\frac2a\Big)^{l+1},
\end{gathered}
\]
where the bound on $\alpha_l$ uses $|\zeta|\ge a/2$ on $N$, and the
substitution $\zeta=\Orot\zeta'$ and the $3$-fold rotational symmetry of $w_i(t,\cdot)$ show
$\alpha_l=0$ unless $3\mid l+1$.
In ring-$i$ units the annulus $N'_k$ lies in $\overline B_{3s_{i,k}}$, where we
evaluate the series. Let $|z|\le3s_{i,k}$, so that $6s_{i,k}/a\le\frac14$. Writing $l=3p-1$ with
$p\ge1$, and using $l(l-1)\cdots(l-q+1)\le(3p)^q$, we get
\[
\begin{aligned}
|F^{(q)}(z)|&\le\frac{\|w_i(t)\|_{L^1}}{2\pi}\sum_{p\ge1}(3p)^q\Big(\frac2a\Big)^{3p}(3s_{i,k})^{3p-1-q}\\
&\le\frac{\|w_i(t)\|_{L^1}}{2\pi}\Big(\frac2a\Big)^{q+1}\Big(\frac{6s_{i,k}}a\Big)^{2-q}C;
\end{aligned}
\]
in the second inequality, the terms with $3p-1<q$ vanish on the left and are
nonnegative on the right, and the others are bounded by means of the identity
\[
(2/a)^{3p}(3s_{i,k})^{3p-1-q}=(2/a)^{q+1}(6s_{i,k}/a)^{2-q}(6s_{i,k}/a)^{3(p-1)}.
\]
Since $6s_{i,k}/a\le\frac14$, the last factor is at most $2^{-3(p-1)}$, and
$(3p)^q\le(3p+q)^q$; this gives the bound by $C$.
In ring-$k$ units,
$(V_i)_{[k]}(z)=\frac{A_ir_i}{A_kr_k}(V_i)_{[i]}(s_{i,k}z)=\mu^{i-k}s_{i,k}^{-1}(V_i)_{[i]}(s_{i,k}z)$,
and $|s_{i,k}z|\le3s_{i,k}$ for $z\in N'$. Hence, on $N'$, by the scaling rule,
Equation~\eqref{eq:BS-complex-deriv} and $\mu^{i-k}s_{i,k}=b_{i,k}$,
\[
|\nabla^q(V_i)_{[k]}(z)|=\mu^{i-k}s_{i,k}^{q-1}2^{q/2}|F^{(q)}(s_{i,k}z)|
\lesssim b_{i,k}\|w_i(t)\|_{L^1},
\]
and summing over $q\le M_h$ gives (4). For the last
assertion, Lemma~\ref{lem:regions-basic}(2), by which the sets $S_i(t)$ are
pairwise disjoint, and the definition of $w_i$ give
$\omega(t)\mathbf 1_{S_1(t)\cup\dots\cup S_{k-1}(t)}=\sum_{i<k}\big(A_i\Phi(\cdot/r_i)+A_iw_i(t,\cdot/r_i)\big)$.
Each $A_i\Phi(\cdot/r_i)$ is radial and supported in $\{|x|\ge ar_i\}$, so it
equals its angular average, and its field vanishes on $B_{ar_i}$ by
Lemma~\ref{lem:hollow}. Moreover
$N'_k\subset\overline B_{3r_k}\subset B_{ar_i}$, because $3r_k=3s_{i,k}r_i<ar_i$.
Summing the bounds over $i<k$ completes the proof.
\end{proof}

The next lemma packages the estimate of Lemma~\ref{lem:transfer}(3) for use in
Step~4 of the bootstrap in Section~\ref{sec:bootstrap}, where it verifies the
inner-field bound required by Theorem~\ref{thm:active}. It makes the $L^1$
input explicit and identifies the estimated field: the velocity induced by
all the inner rings, with vorticity $\omega_{\ge k+1}(t)$, minus their monopole
is precisely $\hin_k$ in ring-$k$ units wherever $\hin_k$ is defined. The
inner vorticity in these units is
$\nu^{\mathrm{in}}_k(t,y)=A_k^{-1}\omega_{\ge k+1}(t,r_ky)$, as defined in
Definition~\ref{def:regions} at physical time $t$. The
same constant $\Cvel$ applies to this bound.

\begin{lemma}
\label{lem:inner-field}
Let $M_h\ge1$ be an integer and use the constant $\Cvel$ of
Lemma~\ref{lem:transfer}. Let $1\le k<n$ and $t\ge0$ be such that
$\omega_{\ge k+1}(t)$ is supported in $\overline B_R$ with $R\le\frac a8r_k$ and
$\|\omega_{\ge k+1}(t)\|_{L^1}\le\Cstar A_{k+1}r_{k+1}^2$. Let $E$ be the
velocity field of $\omega_{\ge k+1}(t)$ minus its monopole. Then
$\|E_{[k]}\|_{C^{M_h}(N')}\le\Cvel\,\mu s_k^2\,(R/r_k)^3$. If moreover
$D_{k+1}(t)\subset B_{ar_k/8}$, then $E_{[k]}=\hin_k(t)$ on $N'$.
\end{lemma}

\begin{proof}
The function $g=\omega_{\ge k+1}(t)$ is $3$-fold
rotationally symmetric by Lemma~\ref{lem:regions-basic}(3), and every $x\in N'_k$ has
$|x|\ge ar_k/4\ge2R$. The computation in the proof of
Lemma~\ref{lem:transfer}(3), with $\|g\|_{L^1}\le\Cstar A_{k+1}r_{k+1}^2$ in
place of $2\pi A_{k+1}r_{k+1}^2$, gives
\[
|\nabla^qE_{[k]}(y)|\lesssim\mu s_k^2(R/r_k)^3\qquad(y\in N',\ 0\le q\le M_h),
\]
with constants depending only on $a$ and $M_h$. Summing over $q$
gives the bound after enlarging the common constant $\Cvel$. Now let
$D_{k+1}(t)\subset B_{ar_k/8}$, so that $\hin_k(t)$ is defined. Since $\KBS$
is homogeneous of degree $-1$, $(\KBS\ast\omega_{\ge k+1}(t))_{[k]}=\KBS\ast\nu^{\mathrm{in}}_k$, and the
monopole of $\omega_{\ge k+1}(t)$ is, in ring-$k$ units,
$\frac1{2\pi}\big(\int\nu^{\mathrm{in}}_k\big)y^\perp/|y|^2$. By
Lemma~\ref{lem:inner-circ}, $\int\omega_{\ge k+1}(t)=\int\omega_{\ge k+1}(0)$, so,
by the definition of $G_k$ (Definition~\ref{def:regions}),
$\int\nu^{\mathrm{in}}_k=A_k^{-1}r_k^{-2}\int\omega_{\ge k+1}(0)=2\pi G_k$. Hence
$E_{[k]}=\KBS\ast\nu^{\mathrm{in}}_k-G_ky^\perp/|y|^2=\hin_k(t)$.
\end{proof}

\subsection{Confinement on a Window}

After ring $k+1$ saturates at the time $T_{k+1}$ of Definition~\ref{def:clock},
it is debris: the bootstrap of Section~\ref{sec:bootstrap} no
longer keeps its support near the annulus $N_{k+1}$. The next proposition shows
that, on a window $[T_{k+1},t_1]$, the material inside ring $k$ nevertheless
stays in the disk $\overline B_{ar_k/8}$, because its speed is at most of
order $A_{k+1}r_{k+1}$; the radius of the disk that contains it grows at most
linearly in the kick time $\tauK$. The bootstrap uses it to keep the inner field
$\hin_k$ defined after $T_{k+1}$ (Lemma~\ref{lem:geometry}\textup{(a)}), and it
bounds the normalized moment $q_{k+1}$, through which the material inside ring
$k$ drives ring $k$ (Section~\ref{sec:kick}). The number $\Lambda r_{k+1}$ in
the proposition is the radius $\varrho(t_1)$ reached at the end of the window.
Condition \textup{(iv)} asks that the disk of this radius lie in
$\overline B_{ar_k/8}$, and that on it the rings $j\le k$, under \textup{(ii)}
and \textup{(iii)}, contribute at most $\Cstar A_{k+1}r_{k+1}$ to the speed.

Here $t_1$ is an endpoint up to which we seek this confinement. In the
bootstrap, it is at most the current time, and the aim is to continue the
control through $T_k$, when ring $k$ saturates
(Lemma~\ref{lem:geometry}\textup{(a)}). The required duration in kick time is
\[
A_{k+1}(T_k-T_{k+1})=\Tpin+\mu\hat T_k.
\]
The later bounds $\ksh_k\asymp s_k^5$ give
$\hat T_k=O(1+\log(1/s_k))$. Thus this interval can be long in kick-time
units; its physical length is divided by $A_{k+1}$. Condition \textup{(iv)}
allows this logarithmic duration: the corresponding $\Lambda$ is
$O(1+\log(1/s_k))$, so both $\Lambda s_k$ and $\Lambda^2s_k$ tend to zero as
$s_k\to0$, with the other parameters fixed.

Assumption \textup{(ii)} keeps ring $k$ and the rings outside it away from
the origin throughout $[T_{k+1},t_1]$, allowing Lemma~\ref{lem:hollow} to
bound their velocity on the inner disk. In the bootstrap these rings are
still quiet or active up to $T_k$, and the stronger annular localization
assumed on the current time interval supplies \textup{(ii)}. The proposition
then supplies confinement of the already saturated inner material on that
same interval. These two controls are combined in the continuity argument
of Section~\ref{sec:bootstrap}.

\begin{proposition}
\label{prop:confine}
Let $1\le k<n$, assume that the clock is defined up to level $k+1$, put
$t_0=T_{k+1}$, and let $t_1\ge t_0$. Put $\tauK=A_{k+1}(t-t_0)$ and
$\Lambda=2+2\Cstar A_{k+1}(t_1-t_0)$. Assume:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $D_{k+1}(t_0)\subset\overline{B}_{2r_{k+1}}$;
\item for every $t\in[t_0,t_1]$ and every $j\le k$,
  $S_j(t)\subset\{|x|\ge\frac a2r_j\}$;
\item $\sbar\le a/16$, $\sbar\le\frac12$ and $\mu^4\sbar^2\le\frac18$;
\item $\Lambda s_k\le a/8$ and
  $\frac83\big(\frac2a\big)^3\Lambda^2s_k\le\mu\Cstar$.
\end{enumerate}
Then for every $t\in[t_0,t_1]$:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $D_{k+1}(t)\subset\overline{B}_{\varrho(t)}$ with
  $\varrho(t)=(2+2\Cstar\tauK)r_{k+1}\le\frac a8r_k$;
\item $|q_{k+1}(t)|\le\Cmom(1+\tauK)^3$, for an absolute constant $\Cmom$.
\end{enumerate}
None of the constants depends on $n$, $k$, $t_0$, $\mu$, $\sbar$, $\delta_0$ or
$\kseed$.
\end{proposition}

\begin{proof}
The constant $\Cstar$ is absolute by Lemma~\ref{lem:speed};
we choose $\Cmom$ below in terms of it. We first bound the
velocity on the disk $\overline B_{\Lambda r_{k+1}}$, and then show by a
continuity argument that $D_{k+1}(t)$ does not leave this disk before $t_1$.

\emph{Speed bound.} Fix $t\in[t_0,t_1]$ and $x\in\overline B_{\Lambda r_{k+1}}$;
by (iv), $|x|\le\Lambda s_kr_k\le\frac a8r_k$. By
Lemma~\ref{lem:regions-basic}(2),(3), $\omega(t)=\omega_{\ge k+1}(t)+\sum_{j\le k}g_j$
with $g_j=\omega(t)\mathbf 1_{S_j(t)}$, so
$v(t,x)=\KBS\ast\omega_{\ge k+1}(t)(x)+\sum_{j\le k}\KBS\ast g_j(x)$, and
$|\KBS\ast\omega_{\ge k+1}(t)(x)|\le\Cstar A_{k+1}r_{k+1}$ by Lemma~\ref{lem:speed}.
It remains to bound the fields of the rings $j\le k$. Let $j\le k$. By Lemma~\ref{lem:regions-basic}(3), $g_j\in C_c^\infty$ is
$3$-fold rotationally symmetric and $\|g_j\|_{L^1}=A_jr_j^2\|\Phi\|_{L^1}\le\pi A_jr_j^2$, since
$0\le\Phi<1$ and $\supp\Phi\subset\overline B_1$. By (ii), $g_j$ is supported in
$\{|y|\ge R_j\}$ with $R_j=\frac a2r_j$, and $|x|\le\frac a8r_k\le R_j/2$.
Lemma~\ref{lem:hollow} with $m_0=3$, $|x|\le\Lambda r_{k+1}$ and
$A_j=\mu^{j-k-1}A_{k+1}$ give
\begin{align*}
|\KBS\ast g_j(x)|&\le\frac4{3\pi}\pi A_jr_j^2|x|^2R_j^{-3}
=\frac43\Big(\frac2a\Big)^3A_jr_j\Big(\frac{|x|}{r_j}\Big)^2\\
&\le\frac43\Big(\frac2a\Big)^3\Lambda^2A_{k+1}r_{k+1}\,\mu^{j-k-1}\frac{r_{k+1}}{r_j}.
\end{align*}
By Equation~\eqref{eq:sk-small}, $r_{k+1}/r_j\le s_k\sbar^{k-j}$, and by (iii),
$\sbar^{k-j}\le2^{j-k}$; with $\mu^{j-k}\le1$,
$\mu^{j-k-1}r_{k+1}/r_j\le\mu^{-1}s_k2^{j-k}$, whose sum over
$j\le k$, a geometric series, is less than $2\mu^{-1}s_k$. By (iv),
$\sum_{j\le k}|\KBS\ast g_j(x)|\le\frac83(\frac2a)^3\Lambda^2s_k\mu^{-1}A_{k+1}r_{k+1}\le\Cstar A_{k+1}r_{k+1}$,
so
\[
|v(t,x)|\le2\Cstar A_{k+1}r_{k+1}\qquad(t\in[t_0,t_1],\ x\in\overline B_{\Lambda r_{k+1}}).\tag{$*$}
\]

\emph{Proof of (a).} By definition $D_{k+1}(t)=\flow_t(D_{k+1}(0))$, and
$D_{k+1}(0)=S_{k+1}\cup\dots\cup S_n$ is compact. Let $t^*$ be the largest
$t\in[t_0,t_1]$ such that $D_{k+1}(t')\subset\overline B_{\Lambda r_{k+1}}$ for
every $t'\in[t_0,t]$; it exists, because the set of such $t$ contains $t_0$, by
(i) and $\Lambda\ge2$, and is closed, because $(t,y)\mapsto\flow_t(y)$ is
continuous (Theorem~\ref{thm:smooth}\textup{(4)}). For $t\in[t_0,t^*]$ and
$y\in D_{k+1}(0)$, the point $\flow_{t'}(y)$ lies in $\overline B_{\Lambda r_{k+1}}$
for $t'\in[t_0,t]$, so \textup{($*$)} gives
$|\partial_{t'}\flow_{t'}(y)|=|v(t',\flow_{t'}(y))|\le2\Cstar A_{k+1}r_{k+1}$.
Integrating from $t_0$ to $t$, and using $|\flow_{t_0}(y)|\le2r_{k+1}$, which
follows from (i), we get
$|\flow_t(y)|\le|\flow_{t_0}(y)|+2\Cstar A_{k+1}r_{k+1}(t-t_0)\le\varrho(t)$. If
$t^*<t_1$, then $\varrho(t^*)<\varrho(t_1)=\Lambda r_{k+1}$, since $\Cstar>0$, and
the uniform continuity of $(t,y)\mapsto|\flow_t(y)|$ on
$[t_0,t_1]\times D_{k+1}(0)$ gives $\delta\in(0,t_1-t^*]$ with
$|\flow_t(y)|<\Lambda r_{k+1}$ for $t\in[t^*,t^*+\delta]$ and $y\in D_{k+1}(0)$,
contradicting the maximality of $t^*$. Hence $t^*=t_1$, and (a) holds, because
$\varrho(t)\le\varrho(t_1)=\Lambda s_kr_k\le\frac a8r_k$ by (iv).

\emph{Proof of (b).} By the definition of $Q_3$
(Definition~I.\ref{I-def:multipole}), (a) and Lemma~\ref{lem:speed},
$|Q_3[\omega_{\ge k+1}(t)]|\le\varrho(t)^3\|\omega_{\ge k+1}(t)\|_{L^1}\le\Cstar A_{k+1}r_{k+1}^2\allowbreak\varrho(t)^3$,
so by the definition of $q_{k+1}$ (Definition~\ref{def:regions}),
$|q_{k+1}(t)|\le\Cstar(\varrho(t)/r_{k+1})^3=\Cstar(2+2\Cstar\tauK)^3$, and
$2+2\Cstar\tauK\lesssim1+\tauK$. Choosing $\Cmom$ large enough proves (b).
\end{proof}

The following definition collects the support conditions under which the
perturbation $w_i$ of every ring that is not debris, and the external field
$h_j$ of every quiet ring, are defined (Definition~\ref{def:regions}), together
with the confinement of
Proposition~\ref{prop:confine}\textup{(a)}.
Lemma~\ref{lem:geometry}\textup{(c)} verifies these conditions in the bootstrap.
Since the clock is constructed recursively, they
must also cover rings whose activation or saturation time has not yet been
assigned.

\begin{definition}
\label{def:loc}
Let $t\ge0$, and consider the times assigned by the recursive clock of
Definition~\ref{def:clock}. A ring $j$ is \textbf{quiet at $t$} if its
activation time $\tsh_j$ has not been assigned, or if it has been assigned
and $t\le\tsh_j$. A ring $i$ is \textbf{debris at $t$} if its saturation time
$T_i$ has been assigned and $t>T_i$. Thus a ring with no assigned saturation
time is included among the rings that are not debris.

The \textbf{localization hypothesis} $(\mathrm{Loc})$ holds at $t$ if both
of the following conditions hold:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item Every ring $i$ that is not debris at $t$ satisfies $S_i(t)\subset N_i$.
\item For every ring $j$ that is quiet at $t$, the outer rings and the inner
  material satisfy
  \[
  S_i(t)\cap B_{6r_j}=\emptyset\quad(i<j),\qquad
  D_{j+1}(t)\subset B_{ar_j/8}.
  \]
  If $T_{j+1}$ has been assigned and $t\ge T_{j+1}$, require in addition
  \[
  D_{j+1}(t)\subset\overline B_{\varrho_j(t)},\qquad
  \varrho_j(t)=\big(2+2\Cstar A_{j+1}(t-T_{j+1})\big)r_{j+1},
  \]
  the confinement radius of Proposition~\ref{prop:confine}\textup{(a)}.
\end{enumerate}
\end{definition}

Under support conditions of this kind the external field $h_j$ of ring $j$ is
small in $\sbar$. The bootstrap uses the next lemma to show that $\eta_j$, the
sum of the norms of $\hin_j$ and $\hout_j$ in $C^{M_h}(N')$, is at most $1$ for every ring $j<n$ that is not debris
(Lemma~\ref{lem:geometry}\textup{(b)}).

\begin{lemma}
\label{lem:field-bound}
Let $M_h\ge1$ be an integer and use the constant $\Cvel$ of
Lemma~\ref{lem:transfer}. Assume
$\sbar\le a/16$ and $\sbar\le\frac12$, let $1\le j<n$ and $t\ge0$, and
assume that $S_i(t)\subset N_i$ for every $i<j$, that $S_i(t)\cap B_{6r_j}=\emptyset$
for every $i<j$, and that $D_{j+1}(t)\subset B_{ar_j/8}$. Then
$\eta_j(t)\le \Cvel\big(\mu\sbar^2+\sbar\big)$.
\end{lemma}

\begin{proof}
All implicit constants in this proof depend only on $a$ and $M_h$, since
$\|\Phi\|_{L^1}\le\pi$. The common constant $\Cvel$ is chosen to dominate
these bounds as well. The hypotheses are those under which
Definition~\ref{def:regions} defines $h_j$ at time $t$, with
$\eta_j(t)=\etin_j(t)+\etout_j(t)$. The inner field is the non-monopole field of
$\omega_{\ge j+1}(t)$ in ring-$j$ units by
Lemma~\ref{lem:inner-field}; the hypotheses of that lemma hold with $R=ar_j/8$, by
Lemma~\ref{lem:speed}. By
Lemma~\ref{lem:transfer}(3) with $R=ar_j/8$,
$\etin_j(t)\lesssim\varepsilon_j(ar_j/(8r_{j+1}))^3=(a/8)^3\mu s_j^2\lesssim\mu\sbar^2$;
here we used $\varepsilon_j=\mu s_j^5$, $r_{j+1}=s_jr_j$ and $s_j\le\sbar$.
If $j=1$, then $\etout_1=0$, as there is no ring outside ring $1$. If $j\ge2$, Lemma~\ref{lem:transfer}(4) with $k=j$ and
Lemma~\ref{lem:regions-basic}(4) give
$\etout_j(t)\lesssim\sum_{i<j}b_{i,j}\|w_i(t)\|_{L^1}\le2\|\Phi\|_{L^1}\sum_{i<j}b_{i,j}$.
By Equation~\eqref{eq:sk-small}, $r_j/r_i\le\sbar^{j-i}$, and $\mu^{i-j}\le1$, so
$b_{i,j}\le\sbar^{j-i}$ and, as $\sbar\le\frac12$,
$\sum_{i<j}b_{i,j}\le\sum_{l\ge1}\sbar^l\le2\sbar$. Hence
$\etout_j(t)\lesssim\sbar$, and adding the two bounds proves
the lemma.
\end{proof}

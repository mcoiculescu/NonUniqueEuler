\section{The Kick}
\label{sec:kick}

We compute the kick that ring $k$ receives from the rings inside it,
for $1\le k<n$. While ring $k$ is quiet, the inner field $\hin_k$ of
Definition~\ref{def:regions} drives its unstable linear response through the
mode-$3$ moment $q_{k+1}$ of the region $D_{k+1}(t)$.
Lemma~\ref{lem:coupling} gives the scalar Equation~\eqref{eq:kick} in the local
kick-time coordinate. Proposition~\ref{prop:kick} bounds the accumulated kick
from above and below; these bounds are used to start ring $k$ in
Section~\ref{sec:relay} (Lemmas~\ref{lem:handoff} and~\ref{lem:kicktrue}).

\begin{lemma}
\label{lem:coupling}
Assume \textup{(N1)}--\textup{(N3)}, fix $n\ge2$ and the objects of
Definitions~\ref{def:regions} and \ref{def:clock}, and let $1\le k<n$.
\begin{enumerate}[label=\textup{(\arabic*)}]
\item Let $t\ge0$ with $D_{k+1}(t)\subset B_{ar_k/8}$. Then
  $-\hin_k(t)\cdot\nabla\Phi$, extended by $0$ outside $N'$, lies in $X^s$
  for every $s\ge0$, and
  \[
  \ellst(G_k)\big(-\hin_k(t)\cdot\nabla\Phi\big)=\mu\,\beta_k\,q_{k+1}(t).
  \]
  Moreover, with $\nu^{\mathrm{in}}_k$ of Definition~\ref{def:regions} at the
  time $t$, the mode-$3$ profile of $\hin_k(t)\cdot\er$ is
  $\frac i2\pout_3[\nu^{\mathrm{in}}_k]\,r^{-4}$ for $a/4\le r\le3$.
\item Assume that the clock is defined up to level $k+1$ and that
  $D_{k+1}(t)\subset B_{ar_k/8}$ for every $t\in[0,\tsh_k]$. Then
  $t\mapsto-\hin_k(t)\cdot\nabla\Phi$ is continuous from $[0,\tsh_k]$ into
  $X^s$ for every $s$, and the physical-time linear response is
  \[
  \wlin_{\mathrm{in},k}(t)=A_k\int_0^t e^{A_k(t-s)\Lin(G_k)}\big(-\hin_k(s)\cdot\nabla\Phi\big)\dd s
  \quad(0\le t\le\tsh_k).
  \]
  For the scalar calculation, use the local coordinate
  $\tauK=A_{k+1}(t-T_{k+1})\in[\tau_0,\Tpin]$, where
  $\tau_0=-A_{k+1}T_{k+1}$, and define
  \[
  \begin{gathered}
  t(\tauK)=T_{k+1}+\tauK/A_{k+1},\qquad
  \tilde q(\tauK)=q_{k+1}(t(\tauK)),\\
  \cin(\tauK)=\ellst(G_k)\big(\wlin_{\mathrm{in},k}(t(\tauK))\big).
  \end{gathered}
  \]
  Then
  $\cin\in C^1([\tau_0,\Tpin])$ and, exactly,
  \begin{equation}
  \label{eq:kick}
  \frac{d\cin}{d\tauK}=\zeta_0\,\cin+\beta_k\,\tilde q,\qquad \zeta_0=\frac{\lambda_k}{\mu},\qquad
  \cin(\tau_0)=0 .
  \end{equation}
  The \textbf{kick amplitude} $\Ain(\tauK)=e^{-\zeta_0\tauK}\cin(\tauK)$ satisfies
  \[
  \cin_k:=\ellst(G_k)\big(\wlin_{\mathrm{in},k}(\tsh_k)\big)=e^{\zeta_0\Tpin}\Ain(\Tpin).
  \]
\item If, in addition to the hypotheses of \textup{(2)}, the rings $i<k$ lie
  outside $B_{6r_k}$ for every $t\in[0,\tsh_k]$, then
  $t\mapsto-\hout_k(t)\cdot\nabla\Phi$ is continuous from $[0,\tsh_k]$ into
  $X^s$ for every $s$, and $\wlin_{\mathrm{out},k}$, defined as
  $\wlin_{\mathrm{in},k}$ with $\hout_k$ in place of $\hin_k$, satisfies $\wlin_{\mathrm{in},k}+\wlin_{\mathrm{out},k}=\wlin$ on
  $[0,\tsh_k]$, where $\wlin$ is the physical-time linear response obtained
  from Theorem~\ref{thm:quiet} with $G=G_k$, using the local coordinate
  $\tau=A_kt$ and field $h(\tau)=h_k(\tau/A_k)$.
\end{enumerate}
\end{lemma}
\begin{proof}
\emph{(1).} This part makes precise the computation made before
Definition~I.\ref{I-def:eigendata}, where $b$ is introduced as the mode-$3$
driving term generated by an interior velocity field of unit outer moment, and
$\beta(G)$ as the component of that term along the unstable eigenfunction. By
Lemma~\ref{lem:regions-basic}(2),(3), the function
$\nu^{\mathrm{in}}_k$ of Definition~\ref{def:regions} at time $t$, the vorticity of
$D_{k+1}(t)$ in ring-$k$ units, is real,
$3$-fold rotationally symmetric, in $C_c^\infty(\R^2)$, and its support avoids $0$; by the
hypothesis $D_{k+1}(t)\subset B_{ar_k/8}$ the support is a compact subset of
$B_{a/8}$, so the largest modulus $\rho$ of a point of the support satisfies $\rho<a/8$. By
Lemma~I.\ref{I-lem:BS-basic}(d), $\KBS\ast\nu^{\mathrm{in}}_k=\nabla^\perp\psi$ with
$\psi=\frac1{2\pi}\log|\cdot|\ast\nu^{\mathrm{in}}_k$ smooth on $\{|y|>\rho\}$.
The monopole $G_ky^\perp/|y|^2$ is orthogonal to $\er$, so the polar form
$v\cdot\er=-r^{-1}\partial_\theta\psi$ of the Biot-Savart law
(Section~I.\ref{I-sec:prelim}) gives $\hin_k(t)\cdot\er=-r^{-1}\partial_\theta\psi$
on $N'$. Put $p_3=\pout_3[\nu^{\mathrm{in}}_k]$, the outer mode-$3$ moment of
Definition~I.\ref{I-def:multipole}. By Lemma~\ref{lem:exterior-mode}, at angular order $3$ and with $\varrho=\rho$, the angular mode $3$ of $\psi$ is
$-\frac16p_3r^{-3}$ for $r>\rho$, and by integration by parts in $\theta$ that of
$\partial_\theta\psi$ is $3i$ times that of $\psi$. Since
$N'=\{a/4\le|y|\le3\}\subset\{|y|>\rho\}$, the mode-$3$ profile of
$\hin_k(t)\cdot\er$ is $\frac i2p_3r^{-4}$ for $a/4\le r\le3$, the last claim
of (1). As $\Phi$ is radial and $\Phi'$ vanishes outside $[a,1]$,
$\hin_k(t)\cdot\nabla\Phi=\Phi'(r)\,\hin_k(t)\cdot\er$, so the mode-$3$
profile of $-\hin_k(t)\cdot\nabla\Phi$ is
$-\frac i2p_3r^{-4}\Phi'(r)=p_3\,b(r)$, with $b$ of Definition~I.\ref{I-def:eigendata}.
By Equation~\eqref{eq:BS-complex}, $\hin_k(t)$ is the restriction to
$N'$ of a field smooth on $\{|y|>\rho\}$, and it is $3$-fold rotationally symmetric because
$\nu^{\mathrm{in}}_k$ is; so $-\hin_k(t)\cdot\nabla\Phi$ is real, smooth,
$3$-fold rotationally symmetric and supported in $\{a\le|y|\le1\}\subset N$, hence in $X^s$
for every $s$. By Definition~I.\ref{I-def:eigendata}, $\ellst(G_k)(w)=n_*(G_k)\,\ellst(w_3)$
depends linearly on the mode-$3$ profile $w_3$ of $w$ only, and
$\beta(G_k)=\ellst(G_k)(b\,e^{3i\theta})=n_*(G_k)\ellst(b)$; so
$\ellst(G_k)(-\hin_k(t)\cdot\nabla\Phi)=p_3\,\beta(G_k)$. It remains to
express $p_3$ through the normalized moment $q_{k+1}(t)$ of
Definition~\ref{def:regions}. Since $\omega_{\ge k+1}(t)$ is real-valued,
the change of variables $x=r_ky$ gives
\[
p_3=\frac1{2\pi A_k}\int\omega_{\ge k+1}(t,r_ky)\,\bar y^3\dd y
=\frac{\overline{Q_3[\omega_{\ge k+1}(t)]}}{2\pi A_kr_k^5}
=\frac{A_{k+1}r_{k+1}^5}{2\pi A_kr_k^5}\,q_{k+1}(t)=\frac{\mu s_k^5}{2\pi}\,q_{k+1}(t),
\]
and (1) follows from $\beta_k=\beta(G_k)s_k^5/(2\pi)$
(Definition~\ref{def:regions}).

\emph{(2).} By the hypothesis and the paragraph after
Lemma~\ref{lem:regions-basic}, $t\mapsto\hin_k(t)$ is continuous from
$[0,\tsh_k]$ into $C^M(N')$ for every integer $M\ge0$. Put
$g(t)=-\hin_k(t)\cdot\nabla\Phi$. By (1), $g(t)\in X^s$ for every $s$,
and by Lemma~I.\ref{I-lem:uniform},
$\|g(t)-g(t')\|_{H^M}\le\Cnab(M)\|\hin_k(t)-\hin_k(t')\|_{C^M(N')}$;
so $g$ is continuous into $X^s$ for every $s$. Since $e^{\theta\Lin(G_k)}$ is a
$C_0$-semigroup on $X^s$ (Lemma~I.\ref{I-lem:semigroup-construct}), the integrand
$s\mapsto e^{A_k(t-s)\Lin(G_k)}g(s)$ is continuous into $X^s$, so
$\wlin_{\mathrm{in},k}$ is well defined.

For the scalar calculation we change from physical time to the local kick
coordinate. Since $A_{k+1}=\mu A_k$ by Equation~\eqref{eq:datum} and
$\tsh_k=T_{k+1}+\Tpin/A_{k+1}$ (Definition~\ref{def:clock}), $\tauK$ runs over
$[\tau_0,\Tpin]$ as $t$ runs over $[0,\tsh_k]$. As $\ellst(G_k)$ is bounded on
$X^0$ (Lemma~I.\ref{I-lem:uniform}), it commutes with the integral defining
$\wlin_{\mathrm{in},k}$. Proposition~I.\ref{I-prop:semigroup} turns the semigroup
under this functional into multiplication by $e^{A_k\lambda_k(t-s)}$.
Using (1), followed by the substitution $r=A_{k+1}(s-T_{k+1})$, gives
\begin{align*}
\cin(\tauK)
&=A_k\int_0^{t(\tauK)}e^{A_k\lambda_k(t(\tauK)-s)}\ellst(G_k)\big(g(s)\big)\dd s\\
&=A_k\mu\beta_k\int_0^{t(\tauK)}e^{A_k\lambda_k(t(\tauK)-s)}q_{k+1}(s)\dd s\\
&=\beta_k\int_{\tau_0}^{\tauK}e^{\zeta_0(\tauK-r)}\tilde q(r)\dd r.
\end{align*}
Here the change of variables uses $\dd s=\dd r/A_{k+1}$ and
$A_k\mu/A_{k+1}=1$.
By Equation~\eqref{eq:omega-ge} and area preservation,
$Q_3[\omega_{\ge k+1}(t)]=\int\sum_{i>k}f^{(n)}_i(x')\,\flow_t(x')^3\dd x'$.
The integrand is supported in the compact set $D_{k+1}(0)$, is bounded uniformly
for $t$ in compact intervals, and is continuous in $t$, as
$(t,x')\mapsto\flow_t(x')$ is continuous (Theorem~\ref{thm:smooth}\textup{(4)}).
Thus $q_{k+1}$, and hence its pullback $\tilde q$, are continuous by dominated
convergence. Taking $e^{\zeta_0\tauK}$ out of the last integral and applying the
fundamental theorem of calculus gives $\cin\in C^1([\tau_0,\Tpin])$,
$\cin(\tau_0)=0$, and Equation~\eqref{eq:kick}. At physical time $t=\tsh_k$,
$\tauK=\Tpin$, so
$\cin_k=\cin(\Tpin)=e^{\zeta_0\Tpin}\Ain(\Tpin)$.

\emph{(3).} By the hypotheses and the paragraph after
Lemma~\ref{lem:regions-basic}, $t\mapsto\hout_k(t)$ is continuous from
$[0,\tsh_k]$ into $C^M(N')$ for every $M$. By Lemma~\ref{lem:regions-basic}(3),
$\nu^{\mathrm{out}}_k$ is real and $3$-fold rotationally symmetric, and, as the
rings $i<k$ lie outside $B_{6r_k}$, it vanishes on $B_6$. Hence
$\hout_k(t)$ is real, $3$-fold rotationally symmetric and smooth on $B_6$, by
Equation~\eqref{eq:BS-complex}, and $B_6$ contains $N'$. As in (1),
$-\hout_k(t)\cdot\nabla\Phi\in X^s$ for every $s$, and this function is
continuous in $t$ into $X^s$ by Lemma~I.\ref{I-lem:uniform}, as in (2).
For the application of Section~\ref{sec:single}, use the local coordinate
$\tau=A_kt$ and the explicit pullback $h(\tau)=h_k(\tau/A_k)$.
The discussion after Lemma~\ref{lem:regions-basic} gives the hypotheses of
Lemma~\ref{lem:reduction} on $[0,A_k\tsh_k]$ with $G=G_k$ and this $h$.
Pulling the resulting linear response back to physical time gives
\[
\wlin(t)=A_k\int_0^t e^{A_k(t-s)\Lin(G_k)}
\big(-h_k(s)\cdot\nabla\Phi\big)\dd s.
\]
Since $h_k=\hin_k+\hout_k$, linearity of this integral proves
$\wlin=\wlin_{\mathrm{in},k}+\wlin_{\mathrm{out},k}$.
\end{proof}

Proposition~\ref{prop:kick} works in the following setting, which isolates
Equation~\eqref{eq:kick}. Let $\zeta_0,\beta\in\C$, put $x=\Re\zeta_0$, let
$\tau_0<\tau_1$ be real, put $I=[\tau_0,\tau_1]$, let
$q\colon I\to\C$ be continuous, let $c$ solve $c'=\zeta_0c+\beta q$ on $I$ with
$c(\tau_0)=0$, and put $A(\tau)=e^{-\zeta_0\tau}c(\tau)$. The factor
$e^{-\zeta_0\tau}$ removes the free growth of $c$, so that $A$ is a weighted
integral of the drive $q$, as the next display shows. Since
$A'=e^{-\zeta_0\tau}(c'-\zeta_0c)=\beta e^{-\zeta_0\tau}q$ and $A(\tau_0)=0$,
\begin{equation}
\label{eq:kick-A}
A(\tau)=\beta\int_{\tau_0}^\tau e^{-\zeta_0s}\,q(s)\dd s\qquad(\tau\in I).
\end{equation}
Under the hypotheses of Lemma~\ref{lem:coupling}\textup{(2)}, the functions
$\cin$, $\tilde q$ and $\Ain$, with $\zeta_0=\lambda_k/\mu$, $\beta=\beta_k$ and
$\tau_1=\Tpin$, are an instance of this setting; here
$\tau_0=-A_{k+1}T_{k+1}<0<\Tpin$ by Lemma~\ref{lem:clock}(1) and
Definition~\ref{def:clock}.

Write $\lamref=\lamst(0)$ and $\sigma=\Re\lamref>0$
(Definition~I.\ref{I-def:eigendata}, Proposition~I.\ref{I-prop:U}(a)).
The reference eigenvalue $\lamref$ is a fixed complex number throughout this
section and the subsequent construction. We also use
$I_p(x)=\int_0^\infty e^{-xs}(1+s)^p\dd s$ for $x>0$
(Equation~(I.\ref{I-eq:Ip})); it decreases as $x$ increases.

The next proposition is an elementary estimate for Equation~\eqref{eq:kick-A}.
Its purpose is to give a quantitative lower bound on the kick, which is used
to start the next ring. The drive is close to an exponential on
$[-T_g,0]$. After the receiving ring's free growth is removed, this gives a
main contribution of size $|\beta\qs|/(|\lamref|(\mu-1))$, because the
modulus of the difference between the two growth exponents is comparable to
$|\lamref|(\mu-1)/\mu$.

The assumptions below make this main contribution survive cancellation.
The length condition \textup{(Kick2)} ensures that the exponential interval
captures a definite part of the accumulated kick. The other three bounds
control, respectively, the later polynomially bounded drive
\textup{(Kick1)}, the error in the exponential approximation
\textup{(Kick3)}, and the earlier history \textup{(Kick4)}. Their contributions
are estimated in absolute value and kept small compared with the main term.

\begin{samepage}
\begin{proposition}
\label{prop:kick}
Assume \textup{(N1)}--\textup{(N3)} and the scalar setting above. Let
$1<\mu\le2$, $G,G'\in[0,\Gone]$, and set
\[
\zeta_0=\frac{\lamst(G)}\mu,\qquad
\lambda_{\mathrm{in}}=\lamst(G'),\qquad
w=\lambda_{\mathrm{in}}-\zeta_0,\qquad x=\Re\zeta_0.
\]
Here $\zeta_0$ is the receiving exponent, $\lambda_{\mathrm{in}}$ is the driving
exponent, and $w$ is their difference.
Assume
\[
\max\big\{|\lamst(G)-\lamref|,|\lambda_{\mathrm{in}}-\lamref|\big\}
\le\frac{\sigma(\mu-1)}{8\mu}.
\]
Let $\tau_0<-T_g<0<\tau_1$, $\bar Q>0$, and let $p\ge0$ be an integer.
Suppose there are $\qs\in\C\setminus\{0\}$ and nonnegative measurable functions
$\err$ on $[-T_g,0]$ and $Q_{\mathrm{pre}}$ on $[\tau_0,-T_g]$ such that
\[
\begin{aligned}
|q(\tau)-\qs e^{\lambda_{\mathrm{in}}\tau}|&\le\err(\tau)
&&(-T_g\le\tau\le0),&&\textup{(Q1)}\\
|q(\tau)|&\le Q_{\mathrm{pre}}(\tau)
&&(\tau_0\le\tau\le-T_g),&&\textup{(Q2)}\\
|q(\tau)|&\le\bar Q(1+\tau)^p
&&(0\le\tau\le\tau_1).&&\textup{(Q3)}
\end{aligned}
\]
Define the weighted errors
\[
\Errg=\int_{-T_g}^0e^{-x\tau}\err(\tau)\dd\tau,\qquad
\Errpre=\int_{\tau_0}^{-T_g}e^{-x\tau}Q_{\mathrm{pre}}(\tau)\dd\tau.
\]
Require the following length and error bounds:
\[
\begin{aligned}
\bar Q\,\Ip{p}{\sigma/4}&\le\frac{|\qs|}{10|\lamref|(\mu-1)},
&&\textup{(Kick1)}\\
e^{-\Re w\,T_g}&\le\tfrac12,
&&\textup{(Kick2)}\\
\max\{\Errg,\Errpre\}&\le\frac{|\qs|}{20|\lamref|(\mu-1)}.
&&\textup{(Kick3)--(Kick4)}
\end{aligned}
\]
Then the accumulated kick satisfies
\[
\frac{|\beta|\,|\qs|}{10|\lamref|(\mu-1)}
\le|A(\tau_1)|\le
\frac{3|\beta|\,|\qs|\,\mu}{|\lamref|(\mu-1)}.
\]
\end{proposition}
\end{samepage}

In Lemma~\ref{lem:kicktrue}, $G'=G_{k+1}$ and $T_g=\hat T_{k+1}$.
For $k+1<n$, the three intervals $[\tau_0,-T_g]$, $[-T_g,0]$ and
$[0,\tau_1]$ describe the quiet and active intervals of ring $k+1$, followed
by the pinning interval of ring $k$, in the local kick coordinate. At the
seeded level $k+1=n$, that lemma uses a slightly shorter exponential
interval.

\begin{proof}
We first compare $w$ with $\lamref$ and bound $x$ from below. Put
$e_\mu=\frac{\mu-1}{8\mu}\le\frac1{16}$ and write $\lambda_{\mathrm{in}}=\lamref+d_1$,
$\lamst(G)=\lamref+d_2$ with $|d_1|,|d_2|\le e_\mu\sigma$. Then
$w=\lamref\frac{\mu-1}\mu+d_1-\frac{d_2}\mu$ with
$|d_1-\frac{d_2}\mu|\le2e_\mu\sigma=\frac{\sigma(\mu-1)}{4\mu}$, so, since
$\Re \lamref=\sigma\le|\lamref|$,
\[
\Re w\ge\frac{\sigma(\mu-1)}\mu-\frac{\sigma(\mu-1)}{4\mu}=\frac34\sigma\frac{\mu-1}\mu>0,\qquad
\frac34\,\frac{|\lamref|(\mu-1)}\mu\le|w|\le\frac54\,\frac{|\lamref|(\mu-1)}\mu .
\]
As $\mu\le2$, $x=\Re\lamst(G)/\mu\ge(1-e_\mu)\sigma/\mu\ge\frac{15}{32}\sigma\ge\frac\sigma4>0$,
and as $\Ip{p}{\cdot}$ is decreasing, $\Ip{p}{x}\le\Ip{p}{\sigma/4}$.

Next we split the integral of Equation~\eqref{eq:kick-A} at $-T_g$ and $0$,
according to the three hypotheses on $q$. Put $\mathsf K=\beta\qs(1-e^{-wT_g})/w$,
the main term, and let $\tau=\tau_1$. Then
\[
A(\tau)=\beta\int_{\tau_0}^{-T_g}e^{-\zeta_0s}q(s)\dd s+\beta\int_{-T_g}^0e^{-\zeta_0s}q(s)\dd s
+\beta\int_0^\tau e^{-\zeta_0s}q(s)\dd s ,
\]
and $|e^{-\zeta_0s}|=e^{-xs}$. By (Q2), the
first term on the right has modulus at most $|\beta|\Errpre$. Since
\[
\int_{-T_g}^0e^{-\zeta_0s}\qs e^{\lambda_{\mathrm{in}}s}\dd s=\qs\int_{-T_g}^0e^{ws}\dd s=\qs(1-e^{-wT_g})/w,
\]
the second term is $\mathsf K$ plus a remainder of modulus at most
$|\beta|\Errg$, by (Q1). By (Q3), $x>0$ and Equation~(I.\ref{I-eq:Ip}), the third
has modulus at most
$|\beta|\bar Q\int_0^\infty e^{-xs}(1+s)^p\dd s=|\beta|\bar Q\,\Ip{p}{x}$.

Finally set $S=|\beta||\qs|/(|\lamref|(\mu-1))$, the natural size of the kick.
By (Kick2) and $|e^{-wT_g}|=e^{-\Re w\,T_g}$, $|1-e^{-wT_g}|\in[\frac12,\frac32]$,
and the bounds on $|w|$ from the first paragraph give
\[
|\mathsf K|\ge\tfrac12\cdot\tfrac45\mu S=\tfrac25\mu S,\qquad
|\mathsf K|\le\tfrac32\cdot\tfrac43\mu S=2\mu S.
\]
By (Kick1) and $\Ip{p}{x}\le\Ip{p}{\sigma/4}$,
$|\beta|\bar Q\,\Ip{p}{x}\le S/10$, and by (Kick3) and (Kick4),
$|\beta|\Errg,|\beta|\Errpre\le S/20$. Thus the three remainders have total
modulus at most $S/5$. The triangle inequality and $\mu>1$ give
\begin{gather*}
|A(\tau)|\ge\big(\tfrac25\mu-\tfrac15\big)S\ge\tfrac15 S\ge\tfrac1{10}S,\\
|A(\tau)|\le\big(2\mu+\tfrac15\big)S\le3\mu S.\qedhere
\end{gather*}
\end{proof}
